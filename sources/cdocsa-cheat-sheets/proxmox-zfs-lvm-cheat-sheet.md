Use whole disk\nzpool create tank /dev/sdb1                   # Use partition\n\n# Mirror (RAID1 equivalent, 2+ disks)\nzpool create tank mirror /dev/sdb /dev/sdc   # 2-way mirror\nzpool create tank mirror /dev/sdb /dev/sdc /dev/sdd  #
